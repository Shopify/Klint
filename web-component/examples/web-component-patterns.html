4. Component Factory Pattern

Create reusable component generators with JavaScript.

// Component factory function
function createSparkline(containerId, data, options = {}) {
    const container = document.getElementById(containerId);
    const klint = document.createElement('klint');
    
    // Set attributes
    klint.setAttribute('width', options.width || '200');
    klint.setAttribute('height', options.height || '50');
    klint.setAttribute('data-values', JSON.stringify(data));
    klint.setAttribute('data-color', options.color || '#007AFF');
    
    // Add the drawing script
    const script = document.createElement('script');
    script.type = 'text/klint';
    script.textContent = `
        function draw(ctx) {
            const element = ctx.canvas.parentElement;
            const values = JSON.parse(element.getAttribute('data-values') || '[]');
            const color = element.getAttribute('data-color');
            
            ctx.background('transparent');
            
            if (values.length > 1) {
                const spacing = ctx.width / (values.length - 1);
                const max = Math.max(...values);
                const min = Math.min(...values);
                const range = max - min || 1;
                
                ctx.strokeColor(color);
                ctx.strokeWidth(2);
                ctx.noFill();
                
                ctx.beginShape();
                values.forEach((val, i) => {
                    const x = i * spacing;
                    const y = ctx.height - ((val - min) / range) * ctx.height;
                    ctx.vertex(x, y);
                });
                ctx.endShape();
            }
        }
    `;
    
    klint.appendChild(script);
    container.appendChild(klint);
    
    return klint;
}

6. Best Practices

Available Attributes

Attribute Type Default Description
width number | "100%" 300 Canvas width in pixels or percentage
height number | "100%" 150 Canvas height in pixels or percentage
fps number 60 Frames per second for animation
static boolean false Render once without animation loop
noloop boolean false Disable automatic animation loop
dpr number | "default" "default" Device pixel ratio
origin "corner" | "center" "corner" Coordinate system origin
alpha boolean true Enable alpha channel
data-* any - Custom data attributes for component configuration

Tips for Reusable Components

  1. Use Custom Data Attributes: Prefix configuration with data- for clean HTML
  2. Read Attributes in Setup: Parse attributes once in setup() for performance
  3. Provide Defaults: Always have fallback values for optional attributes
  4. Keep It Declarative: Embed logic in the script tag when possible
  5. Use Event Delegation: Handle interactions at the component level
  6. Document Your API: Clearly define what attributes your component accepts

<!-- Reusable Progress Ring Component -->
<klint 
    width="150" 
    height="150"
    data-progress="65"
    data-thickness="10"
    data-color="#4ECDC4"
    data-bg-color="#e0e0e0"
    data-show-text="true"
    data-text-format="{value}%">
    <script type="text/klint">
        function setup(ctx) {
            const el = ctx.canvas.parentElement;
            
            // Parse all custom attributes with defaults
            ctx.config = {
                progress: parseFloat(el.getAttribute('data-progress') || 0),
                thickness: parseFloat(el.getAttribute('data-thickness') || 10),
                color: el.getAttribute('data-color') || '#007AFF',
                bgColor: el.getAttribute('data-bg-color') || '#e0e0e0',
                showText: el.getAttribute('data-show-text') !== 'false',
                textFormat: el.getAttribute('data-text-format') || '{value}%'
            };
        }
        
        function draw(ctx) {
            ctx.background('transparent');
            
            const centerX = ctx.width / 2;
            const centerY = ctx.height / 2;
            const radius = Math.min(centerX, centerY) - ctx.config.thickness;
            
            // Background ring
            ctx.strokeColor(ctx.config.bgColor);
            ctx.strokeWidth(ctx.config.thickness);
            ctx.noFill();
            ctx.circle(centerX, centerY, radius);
            
            // Progress ring
            ctx.strokeColor(ctx.config.color);
            ctx.strokeCap('round');
            const angle = (ctx.config.progress / 100) * Math.PI * 2 - Math.PI / 2;
            ctx.arc(centerX, centerY, radius, -Math.PI / 2, angle);
            
            // Text
            if (ctx.config.showText) {
                ctx.fillColor('#333');
                ctx.textSize(24);
                ctx.textWeight('bold');
                ctx.alignText('center', 'middle');
                const text = ctx.config.textFormat.replace('{value}', ctx.config.progress);
                ctx.text(text, centerX, centerY);
            }
        }
    </script>
</klint>
